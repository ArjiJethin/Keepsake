SECTION 1: APPLICATION OVERVIEW & ARCHITECTURAL SUMMARY
================================================================================
Keepsake is an immersive 3D interactive web experience where players explore a low-poly
garden of memories. Built with modern React 19, Three.js, React Three Fiber, and Zustand,
the application combines procedural environmental generation, custom low-poly art, diegetic
parchment UI, and interactive storytelling.

Key Architecture Pillars:
  1. 3D WebGL Scene & Pipeline:
     - Powered by Three.js and @react-three/fiber with @react-three/drei.
     - Dynamic directional lighting, warm sunlight, gentle moonlight, and hemisphere sky fill.
  2. Procedural Infinite Low-Poly Terrain:
     - Chunked terrain system generating deterministic low-poly terrain tiles on the fly.
     - Height calculations driven by customized Simplex noise math and elevation clamping.
     - Procedural distribution of low-poly trees, stones, flowers, and atmospheric dust motes.
  3. Explorer Character Controller & Camera Rig:
     - Custom procedural low-poly 3D explorer mesh (hat, coat, backpack, lantern, animated limbs).
     - Responsive WASD / Arrow movement physics, sprint (Shift), jump (Space), and gravity.
     - Smooth orbital third-person camera with mouse drag/pointer-lock and obstacle avoidance.
  4. Interactive Memory Monuments & Radiance:
     - 3D stone pedestals holding glowing, levitating memory crystals.
     - MemoryRadiance volumetric light motes, nearest-memory telemetry calculation, and proximity detection.
     - Discovered memories trigger journal entries, sound cues, and visual fanfare.
  5. Diegetic UI & Atmospheric Experience:
     - Ink-wash calligraphic loading screen with story prologue.
     - Parchment-themed HUD with compass heading, memory discovery counter, and journal modal.
     - Custom botanical SVG pointer cursor that smoothly follows user interaction.
  6. Centralized Reactive State:
     - Zustand store managing player coordinate stream, discovered memory registry, audio levels,
       and active modal dialogues.

# React + TypeScript + Vite

This template provides a minimal setup to get React working in Vite with HMR and some ESLint rules.

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Oxc](https://oxc.rs)
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/)

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

## Expanding the ESLint configuration

If you are developing a production application, we recommend updating the configuration to enable type-aware lint rules:

```js
export default defineConfig([
  globalIgnores(['dist']),
  {
    files: ['**/*.{ts,tsx}'],
    extends: [
      // Other configs...

      // Remove tseslint.configs.recommended and replace with this
      tseslint.configs.recommendedTypeChecked,
      // Alternatively, use this for stricter rules
      tseslint.configs.strictTypeChecked,
      // Optionally, add this for stylistic rules
      tseslint.configs.stylisticTypeChecked,

      // Other configs...
    ],
    languageOptions: {
      parserOptions: {
        project: ['./tsconfig.node.json', './tsconfig.app.json'],
        tsconfigRootDir: import.meta.dirname,
      },
      // other options...
    },
  },
])

```

// src/store/useGameStore.ts
import { create } from 'zustand';
import type {
    EnvironmentLandmarkZone,
    MemoryItem,
} from '../systems/MemoryManager/memoryTypes';

export type AppPhase = 'loading' | 'intro' | 'exploring';
export type IntroStage = 1 | 2 | 3 | 4 | 5 | 6;
export type QualityPreset = 'high' | 'low';
export type CharacterArchetype = 'female' | 'male';
export type ActiveModal =
    | 'none'
    | 'memory'
    | 'journal'
    | 'friends'
    | 'settings'
    | 'createMemory';

export type InputMode = 'camera' | 'cursor';

export const playerTelemetry = {
    position: [0, 0, 7.5] as [number, number, number],
    velocity: [0, 0, 0] as [number, number, number],
    rotationY: Math.PI,
    cameraYaw: 0,
    cameraPitch: 0.22,
    cameraDistance: 5.2,
    isGrounded: true,
    moveState: 'idle' as 'idle' | 'walk' | 'run' | 'jump',
    nearestMemoryDistance: 999,
    nearestMemoryPosition: [0, 0, 0] as [number, number, number],
    nearestMemoryRadiance: 0, // 0 (normal muted world) -> 1 (full golden vibrancy)
};

interface GameState {
    phase: AppPhase;
    setPhase: (phase: AppPhase) => void;

    introStage: IntroStage;
    setIntroStage: (stage: IntroStage) => void;

    quality: QualityPreset;
    toggleQuality: () => void;

    // Input Mode FSM (Mutually exclusive: camera mode vs cursor mode)
    inputMode: InputMode;
    setInputMode: (mode: InputMode) => void;
    toggleInputMode: () => void;

    // Character selection
    characterModel: CharacterArchetype;
    setCharacterModel: (model: CharacterArchetype) => void;
    toggleCharacterModel: () => void;

    // Terrain generation & ready state
    isInitialTerrainReady: boolean;
    setInitialTerrainReady: (ready: boolean) => void;
    activeChunkCoord: [number, number];
    setActiveChunkCoord: (coord: [number, number]) => void;

    // Persistent memory storage
    persistentMemories: MemoryItem[];
    registerPersistentMemories: (memories: MemoryItem[]) => void;
    advanceMemoryGrowth: (id: string) => void;
    addNewMemory: (memory: MemoryItem) => void;

    // Landmark zones
    landmarkZones: EnvironmentLandmarkZone[];
    nearbyZone: EnvironmentLandmarkZone | null;
    setNearbyZone: (zone: EnvironmentLandmarkZone | null) => void;

    // Memory interaction
    nearbyMemory: MemoryItem | null;
    setNearbyMemory: (memory: MemoryItem | null) => void;

    activeMemory: MemoryItem | null;
    openMemory: (memory: MemoryItem) => void;
    closeMemory: () => void;

    activeEchoId: string | null;
    setActiveEchoId: (id: string | null) => void;

    activeModal: ActiveModal;
    setActiveModal: (modal: ActiveModal) => void;

    discoveredIds: string[];
    markDiscovered: (id: string) => void;
}

export const useGameStore = create<GameState>((set) => ({
    phase: 'loading',
    setPhase: (phase) =>
        set({
            phase,
            // Automatically enter camera mode upon transitioning to exploring
            inputMode: phase === 'exploring' ? 'camera' : 'cursor',
        }),

    introStage: 1,
    setIntroStage: (introStage) => set({ introStage }),

    quality: 'high',
    toggleQuality: () =>
        set((state) => ({
            quality: state.quality === 'high' ? 'low' : 'high',
        })),

    inputMode: 'cursor',
    setInputMode: (inputMode) => set({ inputMode }),
    toggleInputMode: () =>
        set((state) => ({
            inputMode: state.inputMode === 'camera' ? 'cursor' : 'camera',
        })),

    characterModel: 'female',
    setCharacterModel: (characterModel) => set({ characterModel }),
    toggleCharacterModel: () =>
        set((state) => ({
            characterModel: state.characterModel === 'female' ? 'male' : 'female',
        })),

    isInitialTerrainReady: false,
    setInitialTerrainReady: (isInitialTerrainReady) =>
        set({ isInitialTerrainReady }),

    activeChunkCoord: [0, 0],
    setActiveChunkCoord: (coord) =>
        set((state) => {
            if (
                state.activeChunkCoord[0] === coord[0] &&
                state.activeChunkCoord[1] === coord[1]
            ) {
                return state;
            }
            return { activeChunkCoord: coord };
        }),

    persistentMemories: [],
    registerPersistentMemories: (memories) =>
        set((state) => {
            if (state.persistentMemories.length > 0) return state;
            return { persistentMemories: memories };
        }),

    advanceMemoryGrowth: (id) =>
        set((state) => ({
            persistentMemories: state.persistentMemories.map((m) => {
                if (m.id !== id) return m;
                const order: Array<MemoryItem['growthStage']> = [
                    'seed',
                    'plant',
                    'tree',
                    'landmark',
                ];
                const idx = order.indexOf(m.growthStage || 'seed');
                const nextStage = order[Math.min(idx + 1, order.length - 1)];
                return {
                    ...m,
                    growthStage: nextStage,
                    interactionState: 'awakened',
                    isDiscovered: true,
                };
            }),
        })),

    addNewMemory: (memory) =>
        set((state) => ({
            persistentMemories: [memory, ...state.persistentMemories],
            discoveredIds: [...state.discoveredIds, memory.id],
        })),

    landmarkZones: [
        {
            id: 'zone-sakura-kinship',
            name: 'Great Sakura Memory Tree',
            zoneType: 'memory_tree',
            position: [6.2, 0, 2.2],
            radius: 6.0,
            description: 'An ancient blossom tree preserving chapters of shared kinship.',
            associatedCollectionId: 'chapter-kinship',
            unlocked: true,
        },
        {
            id: 'zone-western-arch',
            name: 'Threshold of Hidden Pathways',
            zoneType: 'hidden_pathway',
            position: [-6.8, 0, -3.5],
            radius: 5.2,
            description: 'A moss-draped stone gate leading toward distant memory valleys.',
            associatedCollectionId: 'chapter-journeys',
            unlocked: true,
        },
        {
            id: 'zone-reflection-basin',
            name: 'Stillwater Reflection Glade',
            zoneType: 'reflection_pond',
            position: [-1.5, 0, -14.5],
            radius: 5.5,
            description: 'A quiet hollow where past seasons ripple across the surface.',
            unlocked: true,
        },
    ],
    nearbyZone: null,
    setNearbyZone: (zone) =>
        set((state) => {
            if (state.nearbyZone?.id === zone?.id) return state;
            return { nearbyZone: zone };
        }),

    nearbyMemory: null,
    setNearbyMemory: (memory) =>
        set((state) => {
            if (state.nearbyMemory?.id === memory?.id) return state;
            return { nearbyMemory: memory };
        }),

    activeMemory: null,
    openMemory: (memory) =>
        set((state) => ({
            activeMemory: memory,
            activeEchoId: memory.id,
            activeModal: 'memory',
            inputMode: 'cursor', // Opening memory viewer immediately enables cursor mode
            discoveredIds: state.discoveredIds.includes(memory.id)
                ? state.discoveredIds
                : [...state.discoveredIds, memory.id],
        })),
    closeMemory: () =>
        set({
            activeMemory: null,
            activeEchoId: null,
            activeModal: 'none',
            inputMode: 'camera', // Closing returns directly to camera mode
        }),

    activeEchoId: null,
    setActiveEchoId: (activeEchoId) => set({ activeEchoId }),

    activeModal: 'none',
    setActiveModal: (activeModal) =>
        set({
            activeModal,
            inputMode: activeModal === 'none' ? 'camera' : 'cursor',
        }),

    discoveredIds: [],
    markDiscovered: (id) =>
        set((state) => ({
            discoveredIds: state.discoveredIds.includes(id)
                ? state.discoveredIds
                : [...state.discoveredIds, id],
        })),
}));

// src/systems/Collision/collisionSystem.ts
import * as THREE from 'three';

export interface SolidCollider {
    id: string;
    x: number;
    y: number;
    z: number;
    radius: number;
    height: number;
    blocksCamera: boolean;
}

class SpatialCollisionManager {
    private chunkColliders = new Map<string, SolidCollider[]>();
    private staticColliders: SolidCollider[] = [];

    public setStaticColliders(colliders: SolidCollider[]) {
        this.staticColliders = colliders;
    }

    public registerChunk(chunkKey: string, colliders: SolidCollider[]) {
        this.chunkColliders.set(chunkKey, colliders);
    }

    public unregisterChunk(chunkKey: string) {
        this.chunkColliders.delete(chunkKey);
    }

    /**
     * Resolves horizontal cylinder pushback so the player cannot walk through
     * trees, boulders, archway pillars, or memory pedestals.
     */
    public resolvePlayerMovement(
        nextX: number,
        nextY: number,
        nextZ: number,
        playerRadius = 0.42
    ): [number, number] {
        let resolvedX = nextX;
        let resolvedZ = nextZ;

        const allLists = [this.staticColliders, ...this.chunkColliders.values()];

        for (const list of allLists) {
            for (let i = 0; i < list.length; i++) {
                const col = list[i];
                // Quick vertical overlap check
                if (nextY > col.y + col.height || nextY + 1.7 < col.y) continue;

                const dx = resolvedX - col.x;
                const dz = resolvedZ - col.z;
                const minDist = playerRadius + col.radius;
                const distSq = dx * dx + dz * dz;

                if (distSq < minDist * minDist && distSq > 0.00001) {
                    const dist = Math.sqrt(distSq);
                    const overlap = minDist - dist;
                    resolvedX += (dx / dist) * overlap;
                    resolvedZ += (dz / dist) * overlap;
                }
            }
        }

        return [resolvedX, resolvedZ];
    }

    /**
     * Multi-probe ray vs cylinder + bounding intersection for the Third-Person Camera.
     * Returns the maximum safe fraction [0.12 .. 1.0] along the pivot->camera segment.
     */
    public computeSafeCameraFraction(
        pivot: THREE.Vector3,
        desiredCam: THREE.Vector3,
        cameraMargin = 0.38
    ): number {
        const segDir = desiredCam.clone().sub(pivot);
        const segLen = segDir.length();
        if (segLen < 0.001) return 1.0;
        segDir.divideScalar(segLen);

        let minFraction = 1.0;
        const allLists = [this.staticColliders, ...this.chunkColliders.values()];

        for (const list of allLists) {
            for (let i = 0; i < list.length; i++) {
                const col = list[i];
                if (!col.blocksCamera) continue;

                // 2D ray-circle intersection in XZ plane with vertical height bounds check
                const ox = pivot.x - col.x;
                const oz = pivot.z - col.z;
                const dx = segDir.x;
                const dz = segDir.z;

                const a = dx * dx + dz * dz;
                if (a < 0.0001) continue;

                const r = col.radius + cameraMargin;
                const b = 2 * (ox * dx + oz * dz);
                const c = ox * ox + oz * oz - r * r;
                const disc = b * b - 4 * a * c;

                if (disc >= 0) {
                    const sqrtDisc = Math.sqrt(disc);
                    const t1 = (-b - sqrtDisc) / (2 * a);
                    if (t1 > 0.05 && t1 < segLen) {
                        const hitY = pivot.y + segDir.y * t1;
                        if (hitY >= col.y - 0.3 && hitY <= col.y + col.height + 0.3) {
                            const frac = Math.max(0.15, (t1 - 0.15) / segLen);
                            if (frac < minFraction) {
                                minFraction = frac;
                            }
                        }
                    }
                }
            }
        }

        return minFraction;
    }
}

/**
 * Choreographs the 3D Camera during the Loading/Intro sequence before handing
 * control seamlessly to the ThirdPersonCamera upon entering the garden.
 */
const IntroCameraDirector: React.FC = () => {
    const phase = useGameStore((state) => state.phase);
    const introStage = useGameStore((state) => state.introStage);
    const lookAtTarget = useRef(new THREE.Vector3(-0.6, 5.2, -16));

    useFrame((state, delta) => {
        if (phase !== 'loading') return;

        const t = state.clock.getElapsedTime();
        const pointerX = state.pointer.x * 0.25;
        const pointerY = state.pointer.y * 0.14;

        const isSkyCanvasFrame = introStage <= 2;

        const desiredPos = new THREE.Vector3(
            pointerX,
            isSkyCanvasFrame ? 3.05 : 2.35 + Math.sin(t * 0.45) * 0.04,
            isSkyCanvasFrame ? 11.4 : 12.2
        );

        const desiredTarget = new THREE.Vector3(
            -0.4 + pointerX * 0.6,
            isSkyCanvasFrame ? 5.2 + pointerY : 2.5 + pointerY,
            -16
        );

        state.camera.position.lerp(desiredPos, delta * 1.65);
        lookAtTarget.current.lerp(desiredTarget, delta * 1.65);
        state.camera.lookAt(lookAtTarget.current);
    });

    return null;
};

export const ChunkedTerrain: React.FC = () => {

    const readyChunks = useRef(new Set<string>());

    const handleChunkReady = (key: string) => {
        readyChunks.current.add(key);
        if (readyChunks.current.has('0_0')) {
            setInitialTerrainReady(true);
        }
    };

    useFrame(() => {
        const px = playerTelemetry.position[0];
        const pz = playerTelemetry.position[2];
        const cx = Math.round(px / CHUNK_SIZE);
        const cz = Math.round(pz / CHUNK_SIZE);
        if (cx !== activeChunkCoord[0] || cz !== activeChunkCoord[1]) {
            setActiveChunkCoord([cx, cz]);
        }
    });

    const visibleChunks = useMemo(() => {
        const coords: Array<[number, number]> = [];
        for (let dx = -ACTIVE_CHUNK_RADIUS; dx <= ACTIVE_CHUNK_RADIUS; dx++) {
            for (let dz = -ACTIVE_CHUNK_RADIUS; dz <= ACTIVE_CHUNK_RADIUS; dz++) {
                coords.push([activeChunkCoord[0] + dx, activeChunkCoord[1] + dz]);
            }
        }
        return coords;
    }, [activeChunkCoord]);

    return (
        <group>
            {visibleChunks.map(([cx, cz]) => (
                <TerrainChunk
                    key={`${cx}_${cz}`}
                    cx={cx}
                    cz={cz}
                    onChunkReady={handleChunkReady}
                />
            ))}
        </group>
    );
};

const WorldSpaceBlossomPetals: React.FC = () => {
    const instancedRef = useRef<THREE.InstancedMesh>(null);
    const dummy = useMemo(() => new THREE.Object3D(), []);
    const petalGeo = useMemo(() => new THREE.OctahedronGeometry(1, 0), []);

    const particles = useMemo(
        () =>
            Array.from({ length: 24 }).map((_, i) => ({
                wx: (seededRandom(i * 13 + 1) - 0.5) * 44,
                wy: 1.5 + seededRandom(i * 29 + 2) * 8.5,
                wz: (seededRandom(i * 47 + 3) - 0.5) * 44,
                vx: -0.55 - seededRandom(i * 61) * 0.35,
                vy: -0.62 - seededRandom(i * 73) * 0.28,
                vz: 0.15 + (seededRandom(i * 89) - 0.5) * 0.35,
                phase: seededRandom(i * 97) * Math.PI * 2,
                rotSpeedX: 0.9 + seededRandom(i * 103) * 1.1,
                rotSpeedY: 0.7 + seededRandom(i * 109) * 1.0,
            })),
        []
    );

    useFrame(({ clock }, rawDelta) => {
        if (!instancedRef.current) return;
        const delta = Math.min(rawDelta, 0.05);
        const t = clock.getElapsedTime();
        const [px, , pz] = playerTelemetry.position;
        const maxRadius = 28;

        for (let i = 0; i < particles.length; i++) {
            const p = particles[i];

            p.wx += (p.vx + Math.sin(t * 1.4 + p.phase) * 0.28) * delta;
            p.wy += (p.vy + Math.cos(t * 2.1 + p.phase) * 0.12) * delta;
            p.wz += (p.vz + Math.cos(t * 1.2 + p.phase) * 0.24) * delta;

            const groundY = getGardenTerrainHeight(p.wx, p.wz);

            if (p.wy < groundY + 0.15) {
                p.wy = groundY + 7.5 + seededRandom(i + t) * 3.5;
                p.wx = px + (seededRandom(i * 19 + t) - 0.5) * 38;
                p.wz = pz + (seededRandom(i * 31 + t) - 0.5) * 38;
            }

            const dx = p.wx - px;
            const dz = p.wz - pz;
            if (dx > maxRadius) p.wx -= maxRadius * 2;
            else if (dx < -maxRadius) p.wx += maxRadius * 2;

            if (dz > maxRadius) p.wz -= maxRadius * 2;
            else if (dz < -maxRadius) p.wz += maxRadius * 2;

            dummy.position.set(p.wx, p.wy, p.wz);
            dummy.rotation.set(
                t * p.rotSpeedX + p.phase,
                t * p.rotSpeedY + p.phase,
                Math.sin(t + p.phase) * 0.5
            );
            dummy.scale.set(0.085, 0.022, 0.052);
            dummy.updateMatrix();
            instancedRef.current.setMatrixAt(i, dummy.matrix);
        }

        instancedRef.current.instanceMatrix.needsUpdate = true;
    });

    return (
        <instancedMesh
            ref={instancedRef}
            args={[petalGeo, undefined, particles.length]}
            frustumCulled={false}
        >
            <meshBasicMaterial color="#f5b8b0" transparent opacity={0.86} />
        </instancedMesh>
    );
};

// src/scenes/Environment/lowPolyUtils.ts
import * as THREE from 'three';

export const CHUNK_SIZE = 40;
export const CHUNK_SEGMENTS = 28;
export const ACTIVE_CHUNK_RADIUS = 1; // 3x3 chunk grid = 120m x 120m active around player

export function seededRandom(seed: number): number {
    const x = Math.sin(seed * 127.1 + 311.7) * 43758.5453123;
    return x - Math.floor(x);
}

export function hash2D(ix: number, iz: number, salt = 0): number {
    const dot = ix * 127.1 + iz * 311.7 + salt * 74.7;
    const s = Math.sin(dot) * 43758.5453123;
    return s - Math.floor(s);
}

export function smoothstep(x: number, min: number, max: number): number {
    const t = Math.max(0, Math.min(1, (x - min) / (max - min)));
    return t * t * (3 - 2 * t);
}

/**
 * Continuous 2D Value Noise for seamless cross-chunk terrain generation
 */
export function valueNoise2D(x: number, z: number, salt = 0): number {
    const ix = Math.floor(x);
    const iz = Math.floor(z);
    const fx = x - ix;
    const fz = z - iz;

    const u = fx * fx * (3 - 2 * fx);
    const v = fz * fz * (3 - 2 * fz);

    const a = hash2D(ix, iz, salt);
    const b = hash2D(ix + 1, iz, salt);
    const c = hash2D(ix, iz + 1, salt);
    const d = hash2D(ix + 1, iz + 1, salt);

    return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v;
}

/**
 * Multi-octave Fractal Brownian Motion (fBM)
 */
export function fbm2D(x: number, z: number, octaves = 3): number {
    let value = 0;
    let amplitude = 0.5;
    let frequency = 1;
    for (let i = 0; i < octaves; i++) {
        value += amplitude * (valueNoise2D(x * frequency, z * frequency, i * 17) * 2 - 1);
        frequency *= 2.04;
        amplitude *= 0.5;
    }
    return value;
}

/**
 * Returns the exact X-center of the meandering garden trail for any Z coordinate
 */
export function getPathCenterX(z: number): number {
    return Math.sin(z * 0.075) * 2.4 + Math.cos(z * 0.032) * 1.3 - 0.5;
}

/**
 * Seamless World-Space Terrain Height Function.
 * Because this depends purely on global (x, z), adjacent chunk edges match with
 * 0.00000m error while blending the curated Sanctuary Basin with infinite rolling hills.
 */
export function getGardenTerrainHeight(x: number, z: number): number {
    const pathX = getPathCenterX(z);
    const distFromPath = Math.abs(x - pathX);

    // Smooth flat-bottomed walking valley along the meandering trail
    const pathFlatten = smoothstep(distFromPath, 1.4, 7.5);

    // Sanctuary Basin near origin so the hero Sakura Tree & Stone Archway sit naturally
    const distFromOrigin = Math.sqrt(x * x + z * z);
    const sanctuaryBlend = smoothstep(distFromOrigin, 12, 38);

    // Curated hero knolls near origin (Left archway ridge & Right sakura knoll)
    const leftArchHill =
        x < -3.5 ? smoothstep(Math.abs(x + 3.5), 0, 18) * 1.95 * (1 - sanctuaryBlend * 0.5) : 0;
    const rightSakuraKnoll =
        x > 3.5 ? smoothstep(x - 3.5, 0, 16) * 2.15 * (1 - sanctuaryBlend * 0.5) : 0;

    // Continuous procedural rolling hills & gentle valleys across the infinite world
    const macroHills = fbm2D(x * 0.018, z * 0.018, 3) * 4.6;
    const mesoSlopes = fbm2D(x * 0.055 + 11.3, z * 0.055 - 7.9, 2) * 1.15;
    const microUndulation = Math.sin(x * 0.19 + z * 0.15) * 0.11;

    const rawElevation =
        (macroHills * (0.35 + 0.65 * sanctuaryBlend) +
            mesoSlopes * (0.4 + 0.6 * sanctuaryBlend) +
            leftArchHill +
            rightSakuraKnoll +
            microUndulation) *
        pathFlatten;

    // Slight path depression so flagstones sit nestled in the meadow
    const pathChannel = (1 - pathFlatten) * -0.18;

    return rawElevation + pathChannel - 0.35;
}

/**
 * Computes terrain surface normal via finite differences for slope handling
 */
export function getGardenTerrainNormal(x: number, z: number): THREE.Vector3 {
    const eps = 0.25;
    const hL = getGardenTerrainHeight(x - eps, z);
    const hR = getGardenTerrainHeight(x + eps, z);
    const hD = getGardenTerrainHeight(x, z - eps);
    const hU = getGardenTerrainHeight(x, z + eps);
    return new THREE.Vector3(hL - hR, 2 * eps, hD - hU).normalize();
}

/**
 * Creates an artistically faceted low-poly mesh with welded vertices
 * and intentional flat-shaded facets.
 */
export function createSculptedFacetedGeometry(
    radius: number,
    detail: number,
    scaleXYZ: [number, number, number],
    jitterAmount: number,
    seedOffset: number,
    flattenBottom = false
): THREE.BufferGeometry {
    const baseGeo = new THREE.IcosahedronGeometry(radius, detail);
    const geo = baseGeo.toNonIndexed();
    const pos = geo.attributes.position;

    const vertexMap = new Map<string, [number, number, number]>();

    for (let i = 0; i < pos.count; i++) {
        const vx = pos.getX(i);
        const vy = pos.getY(i);
        const vz = pos.getZ(i);
        const key = `${vx.toFixed(3)}_${vy.toFixed(3)}_${vz.toFixed(3)}`;

        let displaced = vertexMap.get(key);
        if (!displaced) {
            const hash = vx * 19.19 + vy * 37.37 + vz * 73.73 + seedOffset * 13.37;
            const n1 = (seededRandom(hash) - 0.5) * 2 * jitterAmount;
            const n2 = (seededRandom(hash + 7.1) - 0.5) * 2 * jitterAmount;
            const n3 = (seededRandom(hash + 19.3) - 0.5) * 2 * jitterAmount;

            let nx = (vx + n1 * radius) * scaleXYZ[0];
            let ny = (vy + n2 * radius) * scaleXYZ[1];
            let nz = (vz + n3 * radius) * scaleXYZ[2];

            if (flattenBottom && ny < -radius * scaleXYZ[1] * 0.35) {
                ny = -radius * scaleXYZ[1] * 0.38;
            }

            displaced = [nx, ny, nz];
            vertexMap.set(key, displaced);
        }

        pos.setXYZ(i, displaced[0], displaced[1], displaced[2]);
    }

    pos.needsUpdate = true;
    geo.computeVertexNormals();
    return geo;
}

    const currentLookAt = useRef(new THREE.Vector3(0, 1.56, 3.0));
    const currentDistance = useRef(playerTelemetry.cameraDistance);

    // Synchronize Pointer Lock and Canvas Event Listeners
    useEffect(() => {
        const dom = gl.domElement;

        const requestLock = () => {
            if (document.pointerLockElement !== dom) {
                try {
                    dom.requestPointerLock?.();
                } catch {
                    // Browser may defer until user gesture
                }
            }
        };

        const releaseLock = () => {
            if (document.pointerLockElement === dom) {
                try {
                    document.exitPointerLock?.();
                } catch {
                    // Ignore
                }
            }
        };

        // When entering exploring phase in Camera Mode, automatically request lock
        if (phase === 'exploring' && inputMode === 'camera' && activeModal === 'none') {
            requestLock();
        } else {
            releaseLock();
        }

        // Left clicking the 3D canvas while in cursor mode immediately returns to camera mode
        const onCanvasClick = (e: MouseEvent) => {
            const state = useGameStore.getState();
            if (state.phase !== 'exploring' || state.activeModal !== 'none') return;

            if (state.inputMode === 'cursor') {
                e.preventDefault();
                state.setInputMode('camera');
                requestLock();
            } else {
                requestLock();
            }
        };

        // Toggle between Camera Mode & Cursor Mode using Alt or Escape
        const onKeyDown = (e: KeyboardEvent) => {
            if (e.repeat) return;
            const state = useGameStore.getState();
            if (state.phase !== 'exploring') return;

            if (e.key === 'Alt' || e.code === 'AltLeft' || e.code === 'AltRight') {
                e.preventDefault();
                // Toggle input mode cleanly
                state.toggleInputMode();
            } else if (e.key === 'Escape' && state.inputMode === 'camera') {
                e.preventDefault();
                state.setInputMode('cursor');
            }
        };

        const onPointerLockChange = () => {
            const locked = document.pointerLockElement === dom;
            const state = useGameStore.getState();
            // If the browser exited pointer lock via Escape, ensure state switches to cursor mode
            if (!locked && state.inputMode === 'camera' && state.activeModal === 'none') {
                state.setInputMode('cursor');
            }
        };

        // Camera Rotation: Strictly active in MODE A (Camera Mode), frozen in MODE B (Cursor Mode)

            const state = useGameStore.getState();
            if (
                state.phase !== 'exploring' ||
                state.inputMode !== 'camera' ||
                state.activeModal !== 'none'
            ) {
                return; // Zero camera movement while in cursor mode
            }

            const sensitivity = 0.0028;
            const dx = e.movementX;
            const dy = e.movementY;

            // Filter extreme browser pointer attach spikes
            if (Math.abs(dx) > 300 || Math.abs(dy) > 300) return;

            playerTelemetry.cameraYaw -= dx * sensitivity;
            playerTelemetry.cameraPitch = THREE.MathUtils.clamp(
                playerTelemetry.cameraPitch + dy * sensitivity,
                MIN_PITCH,
                MAX_PITCH
            );
        };

        const onWheel = (e: WheelEvent) => {
            const state = useGameStore.getState();
            if (state.phase !== 'exploring' || state.inputMode !== 'camera') return;
            playerTelemetry.cameraDistance = THREE.MathUtils.clamp(
                playerTelemetry.cameraDistance + e.deltaY * 0.004,
                2.6,
                MAX_DISTANCE
            );
        };

        dom.addEventListener('click', onCanvasClick);
        window.addEventListener('keydown', onKeyDown);
        document.addEventListener('pointerlockchange', onPointerLockChange);
        window.addEventListener('mousemove', onMouseMove);
        dom.addEventListener('wheel', onWheel, { passive: true });

        return () => {
            dom.removeEventListener('click', onCanvasClick);
            window.removeEventListener('keydown', onKeyDown);
            document.removeEventListener('pointerlockchange', onPointerLockChange);
            window.removeEventListener('mousemove', onMouseMove);
            dom.removeEventListener('wheel', onWheel);
        };
    }, [gl, phase, inputMode, activeModal, setInputMode]);

    useFrame((state, rawDelta) => {
        if (phase !== 'exploring') return;
        const delta = Math.min(rawDelta, 0.05);

        const [px, py, pz] = playerTelemetry.position;

        // Follow character pivot smoothly
        const desiredPivot = new THREE.Vector3(px, py + 1.56, pz);
        currentPivot.current.lerp(desiredPivot, Math.min(1, delta * 13));

        const yaw = playerTelemetry.cameraYaw;
        const pitch = playerTelemetry.cameraPitch;
        const desiredDist = playerTelemetry.cameraDistance;

        const offsetDir = new THREE.Vector3(
            Math.sin(yaw) * Math.cos(pitch),
            Math.sin(pitch),
            Math.cos(yaw) * Math.cos(pitch)
        ).normalize();

        const idealCamPos = currentPivot.current
            .clone()
            .addScaledVector(offsetDir, desiredDist);

        // Multi-Probe Collision handling (whiskers against obstacles)
        const rightWhisker = new THREE.Vector3(
            Math.cos(yaw),
            0,
            -Math.sin(yaw)
        ).multiplyScalar(0.28);
        const upWhisker = new THREE.Vector3(0, 0.25, 0);

        const fCenter = collisionManager.computeSafeCameraFraction(
            currentPivot.current,
            idealCamPos,
            0.42
        );
        const fLeft = collisionManager.computeSafeCameraFraction(
            currentPivot.current.clone().sub(rightWhisker),
            idealCamPos.clone().sub(rightWhisker),
            0.32
        );
        const fRight = collisionManager.computeSafeCameraFraction(
            currentPivot.current.clone().add(rightWhisker),
            idealCamPos.clone().add(rightWhisker),
            0.32
        );
        const fUp = collisionManager.computeSafeCameraFraction(
            currentPivot.current.clone().add(upWhisker),
            idealCamPos.clone().add(upWhisker),
            0.32
        );

        const safeFraction = Math.min(
            fCenter,
            (fCenter + Math.min(fLeft, fRight, fUp)) * 0.5
        );
        let targetSafeDist = Math.max(MIN_DISTANCE, desiredDist * safeFraction);

        // Prevent ground clipping
        for (const sampleT of [0.35, 0.68, 1.0]) {
            const testPos = currentPivot.current
                .clone()
                .addScaledVector(offsetDir, targetSafeDist * sampleT);
            const groundH = getGardenTerrainHeight(testPos.x, testPos.z) + 0.52;
            if (testPos.y < groundH && sampleT === 1.0) {
                const dyNeeded = groundH - currentPivot.current.y;
                if (offsetDir.y < 0.05) {
                    targetSafeDist = Math.max(MIN_DISTANCE, targetSafeDist * 0.78);
                } else {
                    targetSafeDist = Math.max(
                        MIN_DISTANCE,
                        Math.min(targetSafeDist, dyNeeded / Math.max(0.1, offsetDir.y))
                    );
                }
            }
        }

        const distLerpSpeed =
            targetSafeDist < currentDistance.current ? 18.5 : 4.8;
        currentDistance.current = THREE.MathUtils.lerp(
            currentDistance.current,
            targetSafeDist,
            Math.min(1, delta * distLerpSpeed)
        );

        const finalCamPos = currentPivot.current
            .clone()
            .addScaledVector(offsetDir, currentDistance.current);

        const camGroundFloor =
            getGardenTerrainHeight(finalCamPos.x, finalCamPos.z) + 0.46;
        if (finalCamPos.y < camGroundFloor) {
            finalCamPos.y = camGroundFloor;
        }

        state.camera.position.copy(finalCamPos);

        const forwardAim = new THREE.Vector3(
            -Math.sin(yaw) * 1.45,
            0.22,
            -Math.cos(yaw) * 1.45
        );
        const desiredLook = currentPivot.current.clone().add(forwardAim);
        currentLookAt.current.lerp(desiredLook, Math.min(1, delta * 15));
        state.camera.lookAt(currentLookAt.current);
    });

    return null;
};

export const MemoryPedestals: React.FC = () => {
    const phase = useGameStore((state) => state.phase);
    const activeChunkCoord = useGameStore((state) => state.activeChunkCoord);
    const persistentMemories = useGameStore((state) => state.persistentMemories);
    const registerPersistentMemories = useGameStore(
        (state) => state.registerPersistentMemories
    );
    const nearbyMemory = useGameStore((state) => state.nearbyMemory);
    const setNearbyMemory = useGameStore((state) => state.setNearbyMemory);
    const openMemory = useGameStore((state) => state.openMemory);
    const activeModal = useGameStore((state) => state.activeModal);
    const activeEchoId = useGameStore((state) => state.activeEchoId);
    const landmarkZones = useGameStore((state) => state.landmarkZones);
    const setNearbyZone = useGameStore((state) => state.setNearbyZone);

    const floatGroupRefs = useRef<Record<string, THREE.Group | null>>({});
    const echoWaveRefs = useRef<Record<string, THREE.Mesh | null>>({});
    const lightRefs = useRef<Record<string, THREE.PointLight | null>>({});

    // Initialize persistent memories in global store once
    useEffect(() => {
        const initialized = INITIAL_GARDEN_MEMORIES.map((m) => {
            const y = getGardenTerrainHeight(m.position[0], m.position[2]);
            const cx = Math.round(m.position[0] / CHUNK_SIZE);
            const cz = Math.round(m.position[2] / CHUNK_SIZE);
            return {
                ...m,
                position: [m.position[0], y, m.position[2]] as [number, number, number],
                chunkKey: `${cx}_${cz}`,
            };
        });
        registerPersistentMemories(initialized);
    }, [registerPersistentMemories]);

    // Filter memories to those inside currently loaded terrain chunks (3x3 active grid)
    // Demonstrates true persistence: memories in unloaded chunks unmount cleanly and
    // re-mount with their preserved growthStage & state when the chunk reloads!
    const activeChunkMemories = useMemo(() => {
        const [acx, acz] = activeChunkCoord;
        return persistentMemories.filter((m) => {
            const mcx = Math.round(m.position[0] / CHUNK_SIZE);
            const mcz = Math.round(m.position[2] / CHUNK_SIZE);
            return (
                Math.abs(mcx - acx) <= ACTIVE_CHUNK_RADIUS &&
                Math.abs(mcz - acz) <= ACTIVE_CHUNK_RADIUS
            );
        });
    }, [persistentMemories, activeChunkCoord]);

    // Keyboard 'E' Interaction Listener
    useEffect(() => {
        const onKeyDown = (e: KeyboardEvent) => {
            if (
                e.code === 'KeyE' &&
                phase === 'exploring' &&
                activeModal === 'none'
            ) {
                const currentNearby = useGameStore.getState().nearbyMemory;
                if (currentNearby) {
                    openMemory(currentNearby);
                }
            }
        };
        window.addEventListener('keydown', onKeyDown);
        return () => window.removeEventListener('keydown', onKeyDown);
    }, [phase, activeModal, openMemory]);

    useFrame(({ clock }, delta) => {
        const t = clock.getElapsedTime();
        const [px, , pz] = playerTelemetry.position;

        let closestMemory: MemoryItem | null = null;
        let closestDist = INTERACTION_RADIUS;
        let absoluteNearestDist = 999;

        for (let i = 0; i < activeChunkMemories.length; i++) {
            const mem = activeChunkMemories[i];
            const dx = px - mem.position[0];
            const dz = pz - mem.position[2];
            const dist = Math.hypot(dx, dz);

            if (dist < absoluteNearestDist) {
                absoluteNearestDist = dist;
            }

            if (phase === 'exploring' && dist < closestDist) {
                closestDist = dist;
                closestMemory = mem;
            }

            const isNear = nearbyMemory?.id === mem.id;
            const isEchoing = activeEchoId === mem.id;

            const floatGrp = floatGroupRefs.current[mem.id];
            if (floatGrp) {
                const baseH = mem.id === 'mem-seed-first-step' ? 0.55 : 1.48;
                floatGrp.position.y =
                    baseH + Math.sin(t * (isNear ? 3.0 : 1.7) + i) * 0.08;
                floatGrp.rotation.y += delta * (isEchoing ? 1.4 : isNear ? 0.9 : 0.38);
            }

            const echoWave = echoWaveRefs.current[mem.id];
            if (echoWave) {
                const pulse = ((t * 0.85 + i * 0.3) % 1);
                const s = 1 + pulse * (isEchoing ? 2.4 : 1.35);
                echoWave.scale.set(s, s, 1);
                const mat = echoWave.material as THREE.MeshBasicMaterial;
                mat.opacity = (1 - pulse) * (isEchoing ? 0.75 : isNear ? 0.45 : 0.2);
            }

            const light = lightRefs.current[mem.id];
            if (light) {
                const targetIntensity = isEchoing ? 3.8 : isNear ? 2.9 : 1.45;
                light.intensity = THREE.MathUtils.lerp(
                    light.intensity,
                    targetIntensity,
                    delta * 6
                );
            }
        }

        // Update telemetry so the Explorer's backpack Memory Lantern glows brighter near memories
        playerTelemetry.nearestMemoryDistance = absoluteNearestDist;

        if (phase === 'exploring') {
            setNearbyMemory(closestMemory);

            // Check proximity to Future Interactive Environment Landmark Zones
            let activeZone = null;
            for (const z of landmarkZones) {
                const d = Math.hypot(px - z.position[0], pz - z.position[2]);
                if (d <= z.radius) {
                    activeZone = z;
                    break;
                }
            }
            setNearbyZone(activeZone);
        }
    });

    if (phase !== 'exploring') return null;

    return (
        <group>
            {activeChunkMemories.map((p) => {
                const isNear = nearbyMemory?.id === p.id;
                const isEchoing = activeEchoId === p.id;
                const isOrganicSeedMound =
                    p.id === 'mem-seed-first-step' || p.id === 'mem-distant-horizon-seed';

                return (
                    <group key={p.id} position={p.position}>
                        {isOrganicSeedMound ? (
                            /* Organic Mossy Soil Mound & Pebble Ring for Memory Seeds */
                            <group>
                                <mesh position={[0, 0.14, 0]} castShadow receiveShadow>
                                    <cylinderGeometry args={[0.52, 0.78, 0.28, 8]} />
                                    <meshStandardMaterial
                                        color="#5c6946"
                                        roughness={0.9}
                                        flatShading
                                    />
                                </mesh>
                                <mesh position={[0, 0.28, 0]} castShadow receiveShadow>
                                    <cylinderGeometry args={[0.36, 0.52, 0.12, 8]} />
                                    <meshStandardMaterial
                                        color="#6b5344"
                                        roughness={0.92}
                                        flatShading
                                    />
                                </mesh>
                            </group>
                        ) : (
                            /* Sculpted Low-Poly Stone Pedestal Shrine */
                            <group>
                                <mesh position={[0, 0.24, 0]} castShadow receiveShadow>
                                    <cylinderGeometry args={[1.08, 1.32, 0.48, 7]} />
                                    <meshStandardMaterial
                                        color={THEME.colors.stoneBase}
                                        roughness={0.88}
                                        flatShading
                                    />
                                </mesh>
                                <mesh position={[0, 0.76, 0]} castShadow receiveShadow>
                                    <cylinderGeometry args={[0.78, 0.94, 0.72, 7]} />
                                    <meshStandardMaterial
                                        color={THEME.colors.stoneSunlit}
                                        roughness={0.85}
                                        flatShading
                                    />
                                </mesh>
                                <mesh position={[0, 1.18, 0]} castShadow receiveShadow>
                                    <cylinderGeometry args={[0.9, 0.86, 0.14, 7]} />
                                    <meshStandardMaterial
                                        color={isNear ? '#c9b39b' : THEME.colors.stoneSunlit}
                                        roughness={0.8}
                                        flatShading
                                    />
                                </mesh>
                            </group>
                        )}

                        {/* Ground Acoustic / Memory Ripple Ring */}
                        <mesh
                            ref={(el) => {
                                echoWaveRefs.current[p.id] = el;
                            }}
                            position={[0, isOrganicSeedMound ? 0.36 : 1.27, 0]}
                            rotation={[-Math.PI / 2, 0, 0]}
                        >
                            <ringGeometry args={[0.38, 0.46, 24]} />
                            <meshBasicMaterial
                                color={THEME.colors.memoryGold}
                                transparent
                                opacity={0.35}
                                side={THREE.DoubleSide}
                            />
                        </mesh>

                        {/* Floating Living Memory Object + 3D In-World Echo Projection */}
                        <group
                            ref={(el) => {
                                floatGroupRefs.current[p.id] = el;
                            }}
                            position={[0, isOrganicSeedMound ? 0.55 : 1.48, 0]}
                        >
                            {/* Soft Golden Halo Ring */}
                            <mesh position={[0, 0.22, 0]}>
                                <torusGeometry args={[0.46, 0.018, 8, 32]} />
                                <meshBasicMaterial
                                    color={THEME.colors.memoryGold}
                                    transparent
                                    opacity={isNear || isEchoing ? 0.92 : 0.65}
                                />
                            </mesh>

                            {/* Growth-Stage Specific 3D Mesh ('seed' | 'plant' | 'tree' | 'landmark') */}
                            <MemoryGrowthVisual
                                memory={p}
                                isNear={isNear}
                                isEchoing={isEchoing}
                            />

                                    </mesh>

                                        <planeGeometry args={[0.46, 0.44]} />

                                            color={p.echo?.floatingPhotoPalette?.[0] || '#e8a598'}
                                        />

// src/components/UI/GameHUD.tsx
import React, { useEffect } from 'react';
import {
    BookOpen,
    UserCheck,
    Settings,
    Sparkles,
    X,
    MapPin,
    Calendar,
    Volume2,
    Sprout,
    Compass,
} from 'lucide-react';
import { useGameStore } from '../../store/useGameStore';
import './GameHUD.css';

export const GameHUD: React.FC = () => {
    const phase = useGameStore((state) => state.phase);
    const inputMode = useGameStore((state) => state.inputMode);
    const toggleInputMode = useGameStore((state) => state.toggleInputMode);
    const nearbyMemory = useGameStore((state) => state.nearbyMemory);
    const activeMemory = useGameStore((state) => state.activeMemory);
    const persistentMemories = useGameStore((state) => state.persistentMemories);
    const openMemory = useGameStore((state) => state.openMemory);
    const closeMemory = useGameStore((state) => state.closeMemory);
    const advanceMemoryGrowth = useGameStore((state) => state.advanceMemoryGrowth);
    const discoveredIds = useGameStore((state) => state.discoveredIds);
    const characterModel = useGameStore((state) => state.characterModel);
    const toggleCharacterModel = useGameStore(
        (state) => state.toggleCharacterModel
    );
    const nearbyZone = useGameStore((state) => state.nearbyZone);
    const activeChunkCoord = useGameStore((state) => state.activeChunkCoord);
    const setPhase = useGameStore((state) => state.setPhase);
    const setIntroStage = useGameStore((state) => state.setIntroStage);

    useEffect(() => {
        const onKeyDown = (e: KeyboardEvent) => {
            if (
                e.code === 'KeyC' &&
                phase === 'exploring' &&
                !useGameStore.getState().activeMemory
            ) {
                toggleCharacterModel();
            }
        };
        window.addEventListener('keydown', onKeyDown);
        return () => window.removeEventListener('keydown', onKeyDown);
    }, [phase, toggleCharacterModel]);

    if (phase !== 'exploring') return null;

    const currentMemoryData = activeMemory
        ? persistentMemories.find((m) => m.id === activeMemory.id) || activeMemory
        : null;

    const handleReplayIntro = () => {
        closeMemory();
        setIntroStage(1);
        setPhase('loading');
    };

    return (
        <div className="keepsake-hud-layer">
            {/* Top-Left Keepsake Brand Logo & World Status */}
            <div className="hud-top-left">
                <div
                    className="hud-brand-logo"
                    onClick={handleReplayIntro}
                    title="Click to replay brushstroke intro sequence"
                >
                    <span className="hud-brand-text">Keepsake</span>
                    <svg
                        className="hud-brand-leaf"
                        viewBox="0 0 32 32"
                        width="22"
                        height="22"
                    >
                        <path
                            d="M6,26 C10,12 20,6 30,4 C26,16 18,24 6,26 Z"
                            fill="#f7ecda"
                        />
                    </svg>
                </div>

                <div className="hud-pills-row">
                    <div className="hud-discovery-pill">
                        <span>
                            Memories Awoken: {discoveredIds.length} /{' '}
                            {persistentMemories.length}
                        </span>
                    </div>
                    <div className="hud-discovery-pill hud-chunk-pill">
                        <Compass size={13} />
                        <span>
                            Chunk ({activeChunkCoord[0]}, {activeChunkCoord[1]})
                        </span>
                    </div>
                </div>
            </div>

            {/* Top-Center Landmark Zone Banner */}
            {nearbyZone && !currentMemoryData && (
                <div className="hud-zone-banner">
                    <span className="hud-zone-type">
                        {nearbyZone.zoneType.replace('_', ' ').toUpperCase()}
                    </span>
                    <h3 className="hud-zone-title">{nearbyZone.name}</h3>
                    <p className="hud-zone-desc">{nearbyZone.description}</p>
                </div>
            )}

            {/* Top-Right Action Bar */}
            <div className="hud-top-right">
                <button
                    className="hud-explorer-switch-pill"
                    onClick={toggleCharacterModel}
                    title="Switch Playable Explorer (Shortcut: C)"
                >
                    <UserCheck size={16} />
                    <span>
                        {characterModel === 'female'
                            ? 'Explorer: Female (F)'
                            : 'Explorer: Male (M)'}
                    </span>
                    <kbd className="hud-mini-key">C</kbd>
                </button>

                <button
                    className="hud-circle-btn"
                    onClick={() =>
                        openMemory(nearbyMemory || persistentMemories[0])
                    }
                    title="Open Memory Journal"
                >
                    <BookOpen size={18} />
                </button>

                <button
                    className="hud-circle-btn"
                    onClick={handleReplayIntro}
                    title="Replay Brushstroke Loading Sequence"
                >
                    <Settings size={18} />
                </button>
            </div>

            {/* Contextual Proximity Prompt */}
            {nearbyMemory && !currentMemoryData && (
                <div className="hud-proximity-prompt">
                    <button
                        className="remember-prompt-pill"
                        onClick={() => openMemory(nearbyMemory)}
                    >
                        <span className="remember-prompt-text">Remember?</span>
                        <kbd className="remember-prompt-key">E</kbd>
                    </button>
                    <span className="remember-prompt-subtitle">
                        {nearbyMemory.title} ({nearbyMemory.growthStage?.toUpperCase()})
                    </span>
                </div>
            )}

            {/* Bottom-Left Controls */}
            <div className="hud-bottom-left-controls">
                <div className="control-row">
                    <div className="wasd-cluster">
                        <div className="wasd-top">
                            <kbd className="key-cap key-small">W</kbd>
                        </div>
                        <div className="wasd-bottom">
                            <kbd className="key-cap key-small">A</kbd>
                            <kbd className="key-cap key-small">S</kbd>
                            <kbd className="key-cap key-small">D</kbd>
                        </div>
                    </div>
                    <span className="control-label">Move</span>
                </div>

                <div className="control-row">
                    <span className="mouse-icon-badge" />
                    <span className="control-label">
                        {inputMode === 'camera'
                            ? 'Mode A: Camera Look Active'
                            : 'Mode B: UI Cursor Active (Click World or Alt to Look)'}
                    </span>
                </div>

                <div className="control-row">
                    <kbd
                        className={`key-cap key-wide ${inputMode === 'cursor' ? 'key-cap--active' : ''
                            }`}
                        onClick={toggleInputMode}
                        style={{ cursor: 'pointer', pointerEvents: 'auto' }}
                    >
                        Alt
                    </kbd>
                    <span className="control-label">Toggle Camera / UI Cursor</span>
                </div>

                <div className="control-row">
                    <kbd className="key-cap">E</kbd>
                    <span className="control-label">Awaken Memory Echo</span>
                </div>

                <div className="control-row">
                    <kbd className="key-cap key-wide">Shift</kbd>
                    <span className="control-label">Sprint</span>
                    <kbd className="key-cap key-wide" style={{ marginLeft: 4 }}>
                        Space
                    </kbd>
                    <span className="control-label">Jump</span>
                </div>
            </div>

            {/* Bottom-Right "Remember" Button */}
            <div className="hud-bottom-right">
                <button
                    className="hud-remember-main-btn"
                    onClick={() =>
                        openMemory(nearbyMemory || persistentMemories[0])
                    }
                >
                    <Sparkles size={18} className="remember-flower-icon" />
                    <span>Remember</span>
                </button>
            </div>

            {/* Memory Modal */}
            {currentMemoryData && (
                <div className="memory-modal-backdrop" onClick={closeMemory}>
                    <div
                        className="memory-parchment-card"
                        onClick={(e) => e.stopPropagation()}
                    >
                        <button
                            className="memory-close-btn"
                            onClick={closeMemory}
                            aria-label="Close Memory Echo"
                        >
                            <X size={20} />
                        </button>

                        <div className="memory-card-header">
                            <div className="memory-badges-row">
                                <span className="memory-category-tag">
                                    {currentMemoryData.category.toUpperCase()} ECHO
                                </span>
                                <span className="memory-stage-badge">
                                    <Sprout size={13} />
                                    Stage: {(currentMemoryData.growthStage || 'seed').toUpperCase()}
                                </span>
                                {currentMemoryData.importance && (
                                    <span className="memory-importance-badge">
                                        {currentMemoryData.importance.toUpperCase()}
                                    </span>
                                )}
                            </div>
                            <h2 className="memory-card-title">{currentMemoryData.title}</h2>
                            <p className="memory-card-subtitle">
                                {currentMemoryData.subtitle}
                            </p>
                        </div>

                        <div className="memory-visual-frame">
                            <div className="memory-visual-art">
                                <div className="memory-art-sun" />
                                <div className="memory-art-mountains" />
                                <div className="memory-art-meadow" />
                                {currentMemoryData.echo?.ambientAudioLabel && (
                                    <div className="memory-echo-audio-pill">
                                        <Volume2 size={14} />
                                        <span>{currentMemoryData.echo.ambientAudioLabel}</span>
                                    </div>
                                )}
                            </div>
                        </div>

                        {currentMemoryData.echo?.voiceTranscript && (
                            <blockquote className="memory-voice-echo">
                                {currentMemoryData.echo.voiceTranscript}
                            </blockquote>
                        )}

                        <div className="memory-meta-row">
                            <span className="memory-meta-item">
                                <Calendar size={14} /> {currentMemoryData.date}
                            </span>
                            <span className="memory-meta-item">
                                <MapPin size={14} /> {currentMemoryData.location}
                            </span>
                            {currentMemoryData.people && (
                                <span className="memory-meta-item">
                                    With: {currentMemoryData.people.join(', ')}
                                </span>
                            )}
                        </div>

                        <p className="memory-card-description">
                            {currentMemoryData.description}
                        </p>

                        <div className="memory-tag-list">
                            {currentMemoryData.tags.map((tag) => (
                                <span key={tag} className="memory-tag-chip">
                                    #{tag}
                                </span>
                            ))}
                        </div>

                        <div className="memory-card-footer">
                            {currentMemoryData.growthStage !== 'landmark' && (
                                <button
                                    className="memory-nourish-btn"
                                    onClick={() => advanceMemoryGrowth(currentMemoryData.id)}
                                >
                                    <Sprout size={16} />
                                    <span>Nourish Memory Growth</span>
                                </button>
                            )}
                            <button className="memory-return-btn" onClick={closeMemory}>
                                Return to the Garden
                            </button>
                        </div>
                    </div>
                </div>
            )}
        </div>
    );
};

import React, { useEffect, useRef } from 'react';

import './CustomCursor.css';

export const CustomCursor: React.FC = () => {
    const phase = useGameStore((state) => state.phase);
    const inputMode = useGameStore((state) => state.inputMode);
    const activeModal = useGameStore((state) => state.activeModal);

    const cursorRef = useRef<HTMLDivElement>(null);

    // Custom botanical cursor is only active in Cursor Mode or during UI modals / loading
    const showBotanicalCursor =
        phase !== 'exploring' || inputMode === 'cursor' || activeModal !== 'none';

    useEffect(() => {
        const onMouseMove = (e: MouseEvent) => {
            if (cursorRef.current) {
                cursorRef.current.style.transform = `translate3d(${e.clientX}px, ${e.clientY}px, 0)`;
            }
        };

        window.addEventListener('mousemove', onMouseMove, { passive: true });
        return () => window.removeEventListener('mousemove', onMouseMove);
    }, []);

    useEffect(() => {
        if (showBotanicalCursor) {
            document.body.classList.add('keepsake-custom-cursor-active');
        } else {
            document.body.classList.remove('keepsake-custom-cursor-active');
        }
    }, [showBotanicalCursor]);

    if (!showBotanicalCursor) return null;

    return (
        <div
            ref={cursorRef}
            className={`keepsake-botanical-cursor ${inputMode === 'cursor' ? 'keepsake-botanical-cursor--alt-active' : ''
                }`}
            aria-hidden="true"
        >
            <svg
                width="34"
                height="34"
                viewBox="0 0 36 36"

    useEffect(() => {
        if (phase !== 'loading') return;

        setPaintComplete(false);
        setIntroStage(1);

        let rafId = 0;
        let startTime = 0;
        const PAINT_DURATION_MS = 3200;

        const startTimer = setTimeout(() => {
            setIntroStage(2);

            const pathMetrics = CALLIGRAPHY_GUIDE_STROKES.map((spec, idx) => {
                const el = maskPathRefs.current[idx];
                const len = el ? el.getTotalLength() : 200;
                if (el) {
                    el.style.strokeDasharray = `${len}`;
                    el.style.strokeDashoffset = `${len}`;
                }
                return { el, len, weight: spec.weight };
            });

            const underlineEl = underlineVisRef.current;
            const underlineLen = underlineEl ? underlineEl.getTotalLength() : 480;
            if (underlineEl) {
                underlineEl.style.strokeDasharray = `${underlineLen}`;
                underlineEl.style.strokeDashoffset = `${underlineLen}`;
            }

            let smoothX = 225;
            let smoothY = 46;
            let smoothAngle = -14;

            const animateBrush = (now: number) => {
                if (!startTime) startTime = now;
                const elapsed = now - startTime;
                const rawProgress = Math.min(1, elapsed / PAINT_DURATION_MS);
                const globalT = 0.5 - 0.5 * Math.cos(rawProgress * Math.PI);

                let accumulatedWeight = 0;
                let targetX = smoothX;
                let targetY = smoothY;
                let targetAngle = smoothAngle;

                for (let i = 0; i < pathMetrics.length; i++) {
                    const m = pathMetrics[i];
                    const segStart = accumulatedWeight;
                    const segEnd = accumulatedWeight + m.weight;
                    accumulatedWeight = segEnd;

                    if (!m.el) continue;

                    if (globalT >= segEnd) {
                        m.el.style.strokeDashoffset = '0';
                        if (i === 7 && underlineEl) {
                            underlineEl.style.strokeDashoffset = '0';
                        }
                    } else if (globalT <= segStart) {
                        m.el.style.strokeDashoffset = `${m.len}`;
                    } else {
                        const localRaw = (globalT - segStart) / m.weight;
                        const localEased = localRaw * localRaw * (3 - 2 * localRaw);
                        const drawDist = localEased * m.len;

                        m.el.style.strokeDashoffset = `${m.len - drawDist}`;

                        if (i === 7 && underlineEl) {
                            underlineEl.style.strokeDashoffset = `${underlineLen * (1 - localEased)
                                }`;
                        }

                        const pt = m.el.getPointAtLength(drawDist);
                        const aheadPt = m.el.getPointAtLength(
                            Math.min(m.len, drawDist + 4)
                        );
                        targetX = pt.x;
                        targetY = pt.y;

                        const dy = aheadPt.y - pt.y;
                        if (Math.hypot(dx, dy) > 0.05) {
